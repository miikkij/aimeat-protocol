```javascript
(function (AIMEAT) {
  'use strict';
  var LIB_NAME = 'prhVertailu';
  var MAX_COMPANIES = 5;
  var translations = null;
  var selectedCompanies = [];
  var companyDetails = {};
  var searchResults = [];
  var searchLoading = false;
  var detailsLoading = false;
  var searchError = null;
  var rootContainer = null;

  // --- Helpers ---

  function t(key, trs, vars) {
    if (!key || !trs) return key || '';
    var str = trs[key] != null ? trs[key] : key;
    if (vars && typeof str === 'string') {
      Object.keys(vars).forEach(function (k) {
        str = str.replace('${' + k + '}', vars[k]);
      });
    }
    return str;
  }

  function dv(val) {
    if (val == null) return '-';
    if (typeof val === 'string' || typeof val === 'number' || typeof val === 'boolean') return String(val);
    if (val.value) return val.value;
    if (val.url) return val.url;
    if (val.name) return val.name;
    if (Array.isArray(val)) return val.map(dv).join(', ');
    return JSON.stringify(val);
  }

  function getDescription(descriptions, langCode) {
    if (!descriptions || !Array.isArray(descriptions)) return '-';
    var match = descriptions.find(function (d) { return d.language === langCode; });
    if (match) return match.description || match.value || dv(match);
    if (descriptions.length > 0) return descriptions[0].description || descriptions[0].value || dv(descriptions[0]);
    return '-';
  }

  function escHtml(str) {
    var div = document.createElement('div');
    div.textContent = String(str);
    return div.innerHTML;
  }

  function el(tag, attrs, children) {
    var elem = document.createElement(tag);
    if (attrs) {
      Object.keys(attrs).forEach(function (k) {
        if (k === 'className') {
          elem.className = attrs[k];
        } else if (k.indexOf('on') === 0) {
          elem.addEventListener(k.substring(2).toLowerCase(), attrs[k]);
        } else {
          elem.setAttribute(k, attrs[k]);
        }
      });
    }
    if (children != null) {
      if (typeof children === 'string') {
        elem.textContent = children;
      } else if (Array.isArray(children)) {
        children.forEach(function (c) { if (c) elem.appendChild(c); });
      } else {
        elem.appendChild(children);
      }
    }
    return elem;
  }

  // --- CSS injection ---

      '.prh-vertailu-search-row { display: flex; gap: 0.5rem; margin-bottom: 0.75rem; }',
      '.prh-vertailu-search-input { flex: 1; padding: 0.5rem 0.75rem; border: 1px solid var(--border, #333); border-radius: 6px; background: var(--surface, #1a1a2e); color: var(--text, #e0e0e0); font-size: 0.9rem; outline: none; }',
      '.prh-vertailu-search-input:focus { border-color: var(--primary, #6c63ff); }',

      '.prh-vertailu-search-btn:hover { opacity: 0.9; }',

      '.prh-vertailu-save-btn { padding: 0.5rem 1rem; border: none; border-radius: 6px; background: var(--success, #22c55e); color: #fff; font-size: 0.9rem; cursor: pointer; }',
      '.prh-vertailu-save-btn:hover { opacity: 0.9; }',
      '.prh-vertailu-save-btn:disabled { opacity: 0.5; cursor: not-allowed; }',
      '.prh-vertailu-max-msg { font-size: 0.8rem; color: var(--warning, #f59e0b); margin-bottom: 0.5rem; }'
    ].join('\n');
    document.head.appendChild(style);
  }

  // --- Data access ---

  var dataLib = null;

  function getDataLib() {
    if (!dataLib) {
      dataLib = AIMEAT.prhYritysdata;
      if (!dataLib) throw new Error('prhYritysdata not loaded');
    }
    return dataLib;
  }

  async function loadTranslations() {
    if (translations) return translations;
    try {
      var raw = await AIMEAT.data.getPublic('ext:prh-avoindata', 'i18n.fi');
      translations = raw || {};
    } catch (e) {
      translations = {};
    }
    return translations;
  }

  // --- Company name helper ---

  function getCompanyName(company) {
    if (!company) return '-';
    if (company.names && company.names.length > 0) return company.names[0].name || dv(company.names[0]);
    if (company.name) return company.name;
    return '-';
  }

  function getBusinessId(company) {
    if (!company) return '-';
    if (company.businessId && company.businessId.value) return company.businessId.value;
    if (company.businessId && typeof company.businessId === 'string') return company.businessId;
    return dv(company.businessId);
  }

  function getCompanyForm(company) {
    if (!company || !company.companyForms || company.companyForms.length === 0) return '-';
    var form = company.companyForms[0];
    if (form.descriptions) return getDescription(form.descriptions, 'fi');
    if (form.name) return form.name;
    return dv(form);
  }

  function getMainBusinessLine(company) {
    if (!company || !company.businessLines || company.businessLines.length === 0) return '-';
    var line = company.businessLines[0];
    if (line.descriptions) return getDescription(line.descriptions, 'fi');
    if (line.name) return line.name;
    return dv(line);
  }

  function getRegistrationDate(company) {
    if (!company || !company.registrationDate) return '-';
    return dv(company.registrationDate);
  }

  function getStatus(company) {
    if (!company || !company.companyForms || company.companyForms.length === 0) return '-';
    var form = company.companyForms[0];
    if (form.registrationDate && form.endDate) return t('compare.status.ended', translations) || 'Lopetettu';
    if (form.registrationDate) return t('compare.status.active', translations) || 'Aktiivinen';
    return '-';
  }

  function getAddresses(company) {
    if (!company || !company.addresses || company.addresses.length === 0) return '-';
    var addr = company.addresses[0];
    var parts = [];
    if (addr.street) parts.push(addr.street);
    if (addr.postCode) parts.push(addr.postCode);
    if (addr.city) parts.push(addr.city);
    if (parts.length > 0) return parts.join(', ');
    return dv(addr);
  }

  function getWebsite(company) {
    if (!company || !company.contactDetails || company.contactDetails.length === 0) return '-';
    var web = company.contactDetails.find(function (c) { return c.type === 'Kotisivun www-osoite' || c.type === 'Website'; });
    if (web) return web.value || dv(web);
    return '-';
  }

  // --- Render logic ---

  function renderAll() {
    if (!rootContainer) return;
    rootContainer.innerHTML = '';
    var root = el('div', { className: 'prh-vertailu-root' });

    // Title
    root.appendChild(el('h2', { className: 'prh-vertailu-title' }, t('compare.title', translations) || 'Yritysvertailu'));

    // Search section
    root.appendChild(renderSearchSection());

    // Selected chips
    if (selectedCompanies.length > 0) {
      root.appendChild(renderSelectedChips());
    }

    // Comparison table
    if (selectedCompanies.length >= 2) {
      root.appendChild(renderComparisonTable());
      root.appendChild(renderSaveSection());
    } else if (selectedCompanies.length === 1) {
      var hint = el('div', { className: 'prh-vertailu-empty' }, t('compare.addMore', translations) || 'Lisää vähintään toinen yritys vertailua varten.');
      root.appendChild(hint);
    } else {
      var empty = el('div', { className: 'prh-vertailu-empty' }, t('compare.empty', translations) || 'Hae ja valitse yrityksiä vertailtavaksi.');
      root.appendChild(empty);
    }

    rootContainer.appendChild(root);
  }

  function renderSearchSection() {
    var section = el('div', { className: 'prh-vertailu-section' });

    // Max reached message
    if (selectedCompanies.length >= MAX_COMPANIES) {
      var maxMsg = el('div', { className: 'prh-vertailu-max-msg' },
        t('compare.maxReached', translations, { max: MAX_COMPANIES }) || 'Enimmäismäärä (' + MAX_COMPANIES + ') saavutettu.');
      section.appendChild(maxMsg);
      return section;
    }

    var row = el('div', { className: 'prh-vertailu-search-row' });
    var input = el('input', {
      className: 'prh-vertailu-search-input',
      type: 'text',
      placeholder: t('compare.searchPlaceholder', translations) || 'Hae yritystä nimellä tai Y-tunnuksella...'
    });
    var btn = el('button', {
      className: 'prh-vertailu-search-btn',
      onClick: function () { doSearch(input.value); }
    }, t('compare.searchBtn', translations) || 'Hae');

    input.addEventListener('keydown', function (e) {
      if (e.key === 'Enter') doSearch(input.value);
    });

    if (searchLoading) btn.disabled = true;

    row.appendChild(input);
    row.appendChild(btn);
    section.appendChild(row);

    // Error
    if (searchError) {
      section.appendChild(el('div', { className: 'prh-vertailu-error' }, searchError));
    }

    // Loading
    if (searchLoading) {
      section.appendChild(el('div', { className: 'prh-vertailu-loading' }, t('compare.searching', translations) || 'Haetaan...'));
    }

    // Results
    if (!searchLoading && searchResults.length > 0) {
      var resultList = el('div', { className: 'prh-vertailu-results' });
      searchResults.forEach(function (company) {
        var bid = getBusinessId(company);
        var name = getCompanyName(company);
        var alreadyAdded = selectedCompanies.some(function (s) { return s.businessId === bid; });

        var item = el('div', { className: 'prh-vertailu-result-item' }, [
          el('div', null, [
            el('span', { className: 'prh-vertailu-result-name' }, name),
            el('span', { className: 'prh-vertailu-result-id' }, ' (' + bid + ')')
          ]),
          alreadyAdded
            ? el('span', { className: 'prh-vertailu-result-id' }, t('compare.alreadyAdded', translations) || 'Lisätty')
            : el('span', { className: 'prh-vertailu-result-add' }, '+ ' + (t('compare.add', translations) || 'Lisää'))
        ]);

        if (!alreadyAdded) {
          item.style.cursor = 'pointer';
          item.addEventListener('click', function () { addCompany(bid, name); });
        }

        resultList.appendChild(item);
      });
      section.appendChild(resultList);
    }

    return section;
  }

  function renderSelectedChips() {
    var wrap = el('div', { className: 'prh-vertailu-selected' });
    selectedCompanies.forEach(function (sc) {
      var chip = el('div', { className: 'prh-vertailu-chip' }, [
        el('span', null, sc.name),
        el('span', { className: 'prh-vertailu-chip-remove', onClick: function () { removeCompany(sc.businessId); } }, '\u00d7')
      ]);
      wrap.appendChild(chip);
    });
    return wrap;
  }

  function renderComparisonTable() {
    var section = el('div', { className: 'prh-vertailu-section' });

    if (detailsLoading) {
      section.appendChild(el('div', { className: 'prh-vertailu-loading' }, t('compare.loadingDetails', translations) || 'Ladataan yritystietoja...'));
      return section;
    }

    var companyData = selectedCompanies.map(function (sc) {
      return companyDetails[sc.businessId] || null;
    });

    // Check if all data loaded
    var allLoaded = companyData.every(function (c) { return c !== null; });
    if (!allLoaded) {
      section.appendChild(el('div', { className: 'prh-vertailu-loading' }, t('compare.loadingDetails', translations) || 'Ladataan yritystietoja...'));
      return section;
    }

    var rows = [
      { key: 'companyName', label: t('compare.field.name', translations) || 'Nimi', getter: getCompanyName },
      { key: 'businessId', label: t('compare.field.businessId', translations) || 'Y-tunnus', getter: getBusinessId },
      { key: 'companyForm', label: t('compare.field.companyForm', translations) || 'Yhtiömuoto', getter: getCompanyForm },
      { key: 'businessLine', label: t('compare.field.businessLine', translations) || 'Päätoimiala', getter: getMainBusinessLine },
      { key: 'registrationDate', label: t('compare.field.registrationDate', translations) || 'Rekisteröintipäivä', getter: getRegistrationDate },
      { key: 'status', label: t('compare.field.status', translations) || 'Tila', getter: getStatus },
      { key: 'address', label: t('compare.field.address', translations) || 'Osoite', getter: getAddresses },
      { key: 'website', label: t('compare.field.website', translations) || 'Kotisivu', getter: getWebsite }
    ];

    var tableWrap = el('div', { className: 'prh-vertailu-table-wrap' });
    var table = el('table', { className: 'prh-vertailu-table' });

    // Header row
    var thead = el('thead');
    var headerRow = el('tr');
    headerRow.appendChild(el('th', null, t('compare.field.label', translations) || 'Kenttä'));
    companyData.forEach(function (c) {
      headerRow.appendChild(el('th', null, getCompanyName(c)));
    });
    thead.appendChild(headerRow);
    table.appendChild(thead);

    // Data rows
    var tbody = el('tbody');
    rows.forEach(function (rowDef) {
      var tr = el('tr');
      tr.appendChild(el('th', null, rowDef.label));

      // Collect values to detect differences
      var values = companyData.map(function (c) { return rowDef.getter(c); });
      var allSame = values.every(function (v) { return v === values[0]; });

      values.forEach(function (val) {
        var td = el('td', null, val);
        if (!allSame) {
          td.className = 'prh-vertailu-diff';
        }
        tr.appendChild(td);
      });

      tbody.appendChild(tr);
    });
    table.appendChild(tbody);
    tableWrap.appendChild(table);
    section.appendChild(tableWrap);

    return section;
  }

  function renderSaveSection() {
    var section = el('div', { className: 'prh-vertailu-section' });
    var saveTitle = el('div', { style: 'font-size:0.9rem;margin-bottom:0.35rem;color:var(--text-muted,#888);' },
      t('compare.saveTitle', translations) || 'Tallenna vertailu');

    var row = el('div', { className: 'prh-vertailu-save-row' });
    var titleInput = el('input', {
      className: 'prh-vertailu-save-input',
      type: 'text',
      placeholder: t('compare.savePlaceholder', translations) || 'Anna vertailulle nimi (valinnainen)...'
    });
    var saveBtn = el('button', {
      className: 'prh-vertailu-save-btn',
      onClick: function () { doSave(titleInput.value); }
    }, t('compare.saveBtn', translations) || 'Tallenna');

    row.appendChild(titleInput);
    row.appendChild(saveBtn);
    section.appendChild(saveTitle);
    section.appendChild(row);

    return section;
  }

  // --- Actions ---

  async function doSearch(query) {
    if (!query || query.trim().length < 2) return;
    searchLoading = true;
    searchError = null;
    searchResults = [];
    renderAll();
    try {
      var lib = getDataLib();
      var result = await lib.searchCompanies(query.trim());
      if (Array.isArray(result)) {
        searchResults = result;
      } else if (result && Array.isArray(result.results)) {
        searchResults = result.results;
      } else {
        searchResults = [];
      }
    } catch (e) {
      searchError = t('compare.searchError', translations) || 'Haku epäonnistui: ' + (e.message || e);
    }
    searchLoading = false;
    renderAll();
  }

  async function addCompany(businessId, name) {
    if (selectedCompanies.length >= MAX_COMPANIES) return;
    if (selectedCompanies.some(function (s) { return s.businessId === businessId; })) return;

    selectedCompanies.push({ businessId: businessId, name: name });
    searchResults = [];
    renderAll();

    // Fetch full details
    if (!companyDetails[businessId]) {
      detailsLoading = true;
      renderAll();
      try {
        var lib = getDataLib();
        var detail = await lib.getCompany(businessId);
        companyDetails[businessId] = detail;
      } catch (e) {
        companyDetails[businessId] = { name: name, businessId: { value: businessId }, _error: true };
      }
      detailsLoading = false;
      renderAll();
    }
  }

  function removeCompany(businessId) {
    selectedCompanies = selectedCompanies.filter(function (s) { return s.businessId !== businessId; });
    renderAll();
  }

  async function doSave(title) {
    if (selectedCompanies.length < 2) return;
    try {
      var lib = getDataLib();
      var comparison = {
        title: title || (t('compare.defaultTitle', translations) || 'Vertailu') + ' ' + new Date().toLocaleDateString('fi-FI'),
        companies: selectedCompanies.map(function (sc) {
          return { businessId: sc.businessId, name: sc.name };
        }),
        createdAt: new Date().toISOString()
      };
      await lib.saveComparison(comparison);
      if (AIMEAT['aimeat-ui-dialogs'] && AIMEAT['aimeat-ui-dialogs'].toast) {
        AIMEAT['aimeat-ui-dialogs'].toast(t('compare.saved', translations) || 'Vertailu tallennettu!', 'success');
      }
    } catch (e) {
      if (AIMEAT['aimeat-ui-dialogs'] && AIMEAT['aimeat-ui-dialogs'].toast) {
        AIMEAT['aimeat-ui-dialogs'].toast(t('compare.saveError', translations) || 'Tallennus epäonnistui', 'error');
      }
    }
  }

  // --- Public API ---

  async function render(container) {
    rootContainer = container;
    injectStyles();
    container.innerHTML = '';
    container.appendChild(el('div', { className: 'prh-vertailu-loading' }, t('compare.loading', translations) || 'Ladataan...'));

    try {
      await loadTranslations();
    } catch (e) {
      // translations stay empty, fallback Finnish strings used
    }

    renderAll();
  }

  // Register
  var exports = { render: render };
  if (AIMEAT.register) AIMEAT.register(LIB_NAME, exports);
  AIMEAT[LIB_NAME] = exports;
})(window.AIMEAT || (window.AIMEAT = {}));
```
