/**
 * @file searchCompanies.js
 * @description Hae yrityksiä nimellä tai Y-tunnuksella PRH:n avoimesta YTJ v3 -rajapinnasta.
 */

var BASE_URL = 'https://avoindata.prh.fi/opendata-ytj-api/v3/companies';
var YTUNNUS_PATTERN = /^\d{7}-\d$/;

async function action(ctx) {
  var query = ctx.input.query;

  if (!query || typeof query !== 'string' || query.trim().length === 0) {
    return { error: 'query is required and must be a non-empty string' };
  }

  query = query.trim();

  var url;
  if (YTUNNUS_PATTERN.test(query)) {
    url = BASE_URL + '?businessId=' + encodeURIComponent(query);
  } else {
    url = BASE_URL + '?name=' + encodeURIComponent(query);
  }

  ctx.log.info('Searching PRH API: ' + url);

  try {
    var resp = await ctx.fetch(url, { method: 'GET' });

    if (!resp.ok) {
      ctx.log.error('PRH API returned status ' + resp.status);
      return { error: 'PRH API returned status ' + resp.status, totalResults: 0, companies: [] };
    }

    var data = JSON.parse(resp.text);
    var totalResults = data.totalResults || 0;
    var companies = data.companies || [];

    ctx.log.info('Found ' + totalResults + ' results for query: ' + query);

    return {
      totalResults: totalResults,
      companies: companies
    };
  } catch (err) {
    ctx.log.error('Failed to search PRH API: ' + String(err));
    return { error: 'Failed to search PRH API: ' + String(err), totalResults: 0, companies: [] };
  }
}

// actions/getCompany.js
/**
 * @file getCompany.js
 * @description Hae yksittäisen yrityksen tiedot Y-tunnuksella PRH:n avoimesta rajapinnasta.
 */

var BASE_URL = 'https://avoindata.prh.fi/opendata-ytj-api/v3/companies';

async function action(ctx) {
  var businessId = ctx.input.businessId;

  if (!businessId || typeof businessId !== 'string' || businessId.trim().length === 0) {
    return { error: 'businessId is required' };
  }

  businessId = businessId.trim();

  var url = BASE_URL + '?businessId=' + encodeURIComponent(businessId);

  ctx.log.info('Fetching company: ' + businessId);

  try {
    var resp = await ctx.fetch(url, { method: 'GET' });

    if (!resp.ok) {
      ctx.log.error('PRH API returned status ' + resp.status);
      return { error: 'PRH API returned status ' + resp.status };
    }

    var data = JSON.parse(resp.text);
    var companies = data.companies || [];

    if (companies.length === 0) {
      ctx.log.info('Company not found: ' + businessId);
      return { error: 'Company not found', businessId: businessId };
    }

    var company = companies[0];

    await ctx.memory.set('companies.cache.' + businessId, company);
    ctx.log.info('Cached company data for ' + businessId);

    return { company: company };
  } catch (err) {
    ctx.log.error('Failed to fetch company: ' + String(err));
    return { error: 'Failed to fetch company: ' + String(err) };
  }
}

// actions/addToWatchlist.js
/**
 * @file addToWatchlist.js
 * @description Lisää yritys seurantalistalle extension-muistiin.
 */

async function action(ctx) {
  var businessId = ctx.input.businessId;
  var companyName = ctx.input.companyName;

  if (!businessId || typeof businessId !== 'string' || businessId.trim().length === 0) {
    return { error: 'businessId is required' };
  }
  if (!companyName || typeof companyName !== 'string' || companyName.trim().length === 0) {
    return { error: 'companyName is required' };
  }

  businessId = businessId.trim();
  companyName = companyName.trim();

  var watchlist = await ctx.memory.get('watchlist.items') || [];

  for (var i = 0; i < watchlist.length; i++) {
    if (watchlist[i].businessId === businessId) {
      ctx.log.info('Company already in watchlist: ' + businessId);
      return { error: 'Company already in watchlist', businessId: businessId };
    }
  }

  var newItem = {
    businessId: businessId,
    companyName: companyName,
    addedAt: new Date().toISOString(),
    lastChecked: null,
    lastModified: null
  };

  watchlist.push(newItem);
  await ctx.memory.set('watchlist.items', watchlist);

  ctx.log.info('Added to watchlist: ' + businessId + ' (' + companyName + ')');

  return { added: true, item: newItem, watchlistSize: watchlist.length };
}

// actions/removeFromWatchlist.js
/**
 * @file removeFromWatchlist.js
 * @description Poista yritys seurantalistalta.
 */

async function action(ctx) {
  var businessId = ctx.input.businessId;

  if (!businessId || typeof businessId !== 'string' || businessId.trim().length === 0) {
    return { error: 'businessId is required' };
  }

  businessId = businessId.trim();

  var watchlist = await ctx.memory.get('watchlist.items') || [];
  var originalLength = watchlist.length;

  var filtered = [];
  for (var i = 0; i < watchlist.length; i++) {
    if (watchlist[i].businessId !== businessId) {
      filtered.push(watchlist[i]);
    }
  }

  if (filtered.length === originalLength) {
    ctx.log.info('Company not found in watchlist: ' + businessId);
    return { removed: false, businessId: businessId };
  }

  await ctx.memory.set('watchlist.items', filtered);
  ctx.log.info('Removed from watchlist: ' + businessId);

  return { removed: true, businessId: businessId, watchlistSize: filtered.length };
}

// actions/checkChanges.js
/**
 * @file checkChanges.js
 * @description Tarkista seurattujen yritysten muutokset PRH:n rajapinnasta.
 */

var BASE_URL = 'https://avoindata.prh.fi/opendata-ytj-api/v3/companies';

async function action(ctx) {
  var watchlist = await ctx.memory.get('watchlist.items') || [];

  if (watchlist.length === 0) {
    ctx.log.info('Watchlist is empty, nothing to check');
    return { changesDetected: 0, message: 'Watchlist is empty' };
  }

  var changesLog = await ctx.memory.get('changes.log') || [];
  var changesDetected = 0;
  var now = new Date().toISOString();

  for (var i = 0; i < watchlist.length; i++) {
    var item = watchlist[i];
    var url = BASE_URL + '?businessId=' + encodeURIComponent(item.businessId);

    try {
      var resp = await ctx.fetch(url, { method: 'GET' });

      if (!resp.ok) {
        ctx.log.warn('PRH API returned status ' + resp.status + ' for ' + item.businessId);
        continue;
      }

      var data = JSON.parse(resp.text);
      var companies = data.companies || [];

      if (companies.length === 0) {
        ctx.log.warn('Company not found: ' + item.businessId);
        continue;
      }

      var company = companies[0];
      var currentLastModified = company.lastModified || null;

      if (item.lastModified !== null && currentLastModified !== null && currentLastModified !== item.lastModified) {
        var change = {
          businessId: item.businessId,
          companyName: item.companyName,
          detectedAt: now,
          field: 'lastModified',
          oldValue: item.lastModified,
          newValue: currentLastModified
        };

        changesLog.push(change);
        changesDetected++;

        ctx.log.info('Change detected for ' + item.businessId + ': lastModified ' + item.lastModified + ' -> ' + currentLastModified);
      }

      item.lastChecked = now;
      item.lastModified = currentLastModified;

      await ctx.memory.set('companies.cache.' + item.businessId, company);
    } catch (err) {
      ctx.log.error('Failed to check ' + item.businessId + ': ' + String(err));
    }
  }

  await ctx.memory.set('watchlist.items', watchlist);
  await ctx.memory.set('changes.log', changesLog);

  ctx.log.info('Change check complete. Detected ' + changesDetected + ' changes across ' + watchlist.length + ' companies');

  return {
    changesDetected: changesDetected,
    companiesChecked: watchlist.length,
    checkedAt: now
  };
}

// actions/init.js
/**
 * @file init.js
 * @description Alusta laajennuksen tiedot ja kopioi jaetut asetukset omistajan nimiavaruudesta.
 */

async function action(ctx) {
  var keysToSync = ['i18n.fi', 'i18n.en', 'settings.config'];
  var prefix = 'prh-yritystietopalvelu.';
  var synced = [];

  for (var i = 0; i < keysToSync.length; i++) {
    var key = keysToSync[i];
    try {
      var value = await ctx.memory.getPublic(ctx.caller.gaii, prefix + key);
      if (value !== null && value !== undefined) {
        await ctx.memory.set('shared.' + key, value);
        synced.push(key);
        ctx.log.info('Synced shared data: ' + key);
      }
    } catch (err) {
      ctx.log.warn('Could not sync key ' + key + ': ' + String(err));
    }
  }

  var watchlist = await ctx.memory.get('watchlist.items');
  if (watchlist === null || watchlist === undefined) {
    await ctx.memory.set('watchlist.items', []);
    ctx.log.info('Initialized watchlist.items');
  }

  var changesLog = await ctx.memory.get('changes.log');
  if (changesLog === null || changesLog === undefined) {
    await ctx.memory.set('changes.log', []);
    ctx.log.info('Initialized changes.log');
  }

  var comparisons = await ctx.memory.get('comparisons.saved');
  if (comparisons === null || comparisons === undefined) {
    await ctx.memory.set('comparisons.saved', []);
    ctx.log.info('Initialized comparisons.saved');
  }

  ctx.log.info('Extension initialized. Synced ' + synced.length + ' shared keys.');

  return {
    initialized: true,
    syncedKeys: synced,
    dataStores: ['watchlist.items', 'changes.log', 'comparisons.saved']
  };
}
