(function (AIMEAT) {
  'use strict';

  // ── Translation helper ──────────────────────────────────────────────
  function t(key, translations, vars) {
    if (!translations) return key;
    // flat key first
    var val = translations[key];
    // dot-path fallback
    if (val === undefined) {
      var parts = key.split('.');
      val = translations;
      for (var i = 0; i < parts.length; i++) {
        if (val == null) break;
        val = val[parts[i]];
      }
    }
    if (typeof val !== 'string') return key;
    if (vars) {
      Object.keys(vars).forEach(function (k) {
        val = val.split('{{' + k + '}}').join(vars[k]);
      });
    }
    return val;
  }

  // ── Display value helper ────────────────────────────────────────────
  function dv(val) {
    if (val == null) return '';
    if (typeof val === 'object') {
      if (val.value !== undefined) return String(val.value);
      if (Array.isArray(val)) {
        if (val.length === 0) return '';
        if (val[0] && val[0].name) return val[0].name;
        return val.map(function (v) { return dv(v); }).join(', ');
      }
      return JSON.stringify(val);
    }
    return String(val);
  }

  // ── Description by language ─────────────────────────────────────────
  function getDescription(descriptions, langCode) {
    if (!descriptions || !Array.isArray(descriptions) || descriptions.length === 0) return '';
    var match = descriptions.find(function (d) { return d.languageCode === langCode; });
    if (match) return match.description || '';
    // fallback: first available
    return descriptions[0].description || '';
  }

  // ── Date formatter ──────────────────────────────────────────────────
  function fmtDate(dateStr) {
    if (!dateStr) return '';
    try {
      var d = new Date(dateStr);
      return d.toLocaleDateString('fi-FI');
    } catch (e) {
      return String(dateStr);
    }
  }

  // ── DOM helpers ─────────────────────────────────────────────────────
  function el(tag, className, textContent) {
    var e = document.createElement(tag);
    if (className) e.className = className;
    if (textContent !== undefined) e.textContent = textContent;
    return e;
  }

  // ── Main render ─────────────────────────────────────────────────────
  function render(container) {
    injectStyles();
    var translations = null;
    var state = { view: 'search', results: null, query: '', detail: null, loading: false, error: null };

    var root = el('div', 'prh-haku-root');
    container.innerHTML = '';
    container.appendChild(root);

    // Load translations
    (async function () {
      try {
        translations = await AIMEAT.data.getPublic('ext:prh-avoindata', 'i18n.fi');
      } catch (e) {
        // translations stay null, keys shown as-is
      }
      draw();
    })();

    function draw() {
      root.innerHTML = '';
      if (state.view === 'search') {
        drawSearch();
      } else if (state.view === 'detail') {
        drawDetail();
      }
    }

    // ── Search view ───────────────────────────────────────────────────
    function drawSearch() {
      // Search bar
      var bar = el('div', 'prh-haku-search');
      var input = el('input', 'prh-haku-input');
      input.type = 'text';
      input.placeholder = t('search.placeholder', translations) !== 'search.placeholder'
        ? t('search.placeholder', translations)
        : 'Yrityksen nimi tai Y-tunnus...';
      input.value = state.query;
      input.addEventListener('keydown', function (e) {
        if (e.key === 'Enter') doSearch(input.value);
      });

      var btn = el('button', 'prh-haku-btn', t('search.button', translations) !== 'search.button'
        ? t('search.button', translations)
        : 'Hae');
      btn.addEventListener('click', function () { doSearch(input.value); });
      if (state.loading) btn.disabled = true;

      bar.appendChild(input);
      bar.appendChild(btn);
      root.appendChild(bar);

      // Error
      if (state.error) {
        var errDiv = el('div', 'prh-haku-error', state.error);
        root.appendChild(errDiv);
      }

      // Loading
      if (state.loading) {
        var loadDiv = el('div', 'prh-haku-status', t('search.loading', translations) !== 'search.loading'
          ? t('search.loading', translations)
          : 'Haetaan...');
        root.appendChild(loadDiv);
        return;
      }

      // No query yet
      if (!state.results) {
        var hint = el('div', 'prh-haku-status', t('search.hint', translations) !== 'search.hint'
          ? t('search.hint', translations)
          : 'Syota yrityksen nimi tai Y-tunnus aloittaaksesi haun.');
        root.appendChild(hint);
        return;
      }

      // Empty results
      if (state.results.length === 0) {
        var empty = el('div', 'prh-haku-status', t('search.noResults', translations) !== 'search.noResults'
          ? t('search.noResults', translations)
          : 'Ei hakutuloksia.');
        root.appendChild(empty);
        return;
      }

      // Result count
      var countText = t('search.resultCount', translations, { count: state.results.length });
      if (countText === 'search.resultCount') {
        countText = state.results.length + ' tulosta';
      }
      var countDiv = el('div', 'prh-haku-count', countText);
      root.appendChild(countDiv);

      // Result cards
      var list = el('div', 'prh-haku-results');
      state.results.forEach(function (company) {
        var card = el('div', 'prh-haku-card');

        var nameDiv = el('div', 'prh-haku-card-name');
        // company.names is an array, get the current name
        var companyName = '';
        if (company.names && company.names.length > 0) {
          companyName = company.names[0].name || '';
        } else if (company.name) {
          companyName = company.name;
        }
        nameDiv.textContent = companyName;
        card.appendChild(nameDiv);

        var meta = el('div', 'prh-haku-card-meta');

        // Business ID
        var bidSpan = el('span');
        bidSpan.textContent = dv(company.businessId);
        meta.appendChild(bidSpan);

        // Company form
        if (company.companyForms && company.companyForms.length > 0) {
          var formSpan = el('span');
          var formDesc = getDescription(company.companyForms[0].descriptions, 1);
          formSpan.textContent = formDesc || company.companyForms[0].type || '';
          meta.appendChild(formSpan);
        }

        // Main business line
        if (company.mainBusinessLines && company.mainBusinessLines.length > 0) {
          var lineSpan = el('span');
          var lineDesc = getDescription(company.mainBusinessLines[0].descriptions, 1);
          lineSpan.textContent = lineDesc || '';
          meta.appendChild(lineSpan);
        }

        card.appendChild(meta);

        card.addEventListener('click', function () {
          loadCompanyDetail(dv(company.businessId), companyName);
        });

        list.appendChild(card);
      });
      root.appendChild(list);
    }

    // ── Detail view ───────────────────────────────────────────────────
    function drawDetail() {
      var wrapper = el('div', 'prh-haku-detail');

      // Back button
      var backBtn = el('button', 'prh-haku-back');
      backBtn.innerHTML = '&#8592; ' + (t('detail.back', translations) !== 'detail.back'
        ? t('detail.back', translations)
        : 'Takaisin hakutuloksiin');
      backBtn.addEventListener('click', function () {
        state.view = 'search';
        draw();
      });
      wrapper.appendChild(backBtn);

      // Loading
      if (state.loading) {
        var loadDiv = el('div', 'prh-haku-status', t('detail.loading', translations) !== 'detail.loading'
          ? t('detail.loading', translations)
          : 'Ladataan yritystietoja...');
        wrapper.appendChild(loadDiv);
        root.appendChild(wrapper);
        return;
      }

      // Error
      if (state.error) {
        var errDiv = el('div', 'prh-haku-error', state.error);
        wrapper.appendChild(errDiv);
        root.appendChild(wrapper);
        return;
      }

      var c = state.detail;
      if (!c) return;

      // Header
      var header = el('div', 'prh-haku-detail-header');
      var companyName = '';
      if (c.names && c.names.length > 0) {
        companyName = c.names[0].name || '';
      } else if (c.name) {
        companyName = c.name;
      }
      var h2 = el('h2', '', companyName);
      header.appendChild(h2);
      var sub = el('div', 'prh-haku-subtitle', dv(c.businessId));
      header.appendChild(sub);
      wrapper.appendChild(header);

      // ── Basic info section ──
      var basicSection = el('div', 'prh-haku-section');
      var basicTitle = el('div', 'prh-haku-section-title',
        t('detail.basicInfo', translations) !== 'detail.basicInfo'
          ? t('detail.basicInfo', translations) : 'Perustiedot');
      basicSection.appendChild(basicTitle);

      var dl = el('dl', 'prh-haku-dl');
      var basicFields = [
        { label: t('detail.businessId', translations) !== 'detail.businessId' ? t('detail.businessId', translations) : 'Y-tunnus', value: dv(c.businessId) },
        { label: t('detail.euId', translations) !== 'detail.euId' ? t('detail.euId', translations) : 'EU-tunnus', value: c.euIdNumbers && c.euIdNumbers.length > 0 ? dv(c.euIdNumbers[0]) : '' },
        { label: t('detail.registrationDate', translations) !== 'detail.registrationDate' ? t('detail.registrationDate', translations) : 'Rekisteroitymispaiva', value: c.businessId && c.businessId.registrationDate ? fmtDate(c.businessId.registrationDate) : '' },
        { label: t('detail.lastModified', translations) !== 'detail.lastModified' ? t('detail.lastModified', translations) : 'Viimeksi muokattu', value: c.lastModified ? fmtDate(c.lastModified) : '' }
      ];
      basicFields.forEach(function (f) {
        if (!f.value) return;
        var dt = el('dt', '', f.label);
        var dd = el('dd', '', f.value);
        dl.appendChild(dt);
        dl.appendChild(dd);
      });
      basicSection.appendChild(dl);
      wrapper.appendChild(basicSection);

      // ── Company form section ──
      if (c.companyForms && c.companyForms.length > 0) {
        var formSection = el('div', 'prh-haku-section');
        var formTitle = el('div', 'prh-haku-section-title',
          t('detail.companyForm', translations) !== 'detail.companyForm'
            ? t('detail.companyForm', translations) : 'Yhtiolomake');
        formSection.appendChild(formTitle);

        var formDl = el('dl', 'prh-haku-dl');
        var form = c.companyForms[0];
        var formType = el('dt', '', t('detail.type', translations) !== 'detail.type' ? t('detail.type', translations) : 'Tyyppi');
        var formTypeVal = el('dd', '', form.type || '');
        formDl.appendChild(formType);
        formDl.appendChild(formTypeVal);

        var formDescLabel = el('dt', '', t('detail.description', translations) !== 'detail.description' ? t('detail.description', translations) : 'Kuvaus');
        var formDescVal = el('dd', '', getDescription(form.descriptions, 1));
        formDl.appendChild(formDescLabel);
        formDl.appendChild(formDescVal);

        if (form.registrationDate) {
          var fRegDt = el('dt', '', t('detail.registrationDate', translations) !== 'detail.registrationDate' ? t('detail.registrationDate', translations) : 'Rekisteroitymispaiva');
          var fRegDd = el('dd', '', fmtDate(form.registrationDate));
          formDl.appendChild(fRegDt);
          formDl.appendChild(fRegDd);
        }

        formSection.appendChild(formDl);
        wrapper.appendChild(formSection);
      }

      // ── Main business line section ──
      if (c.mainBusinessLines && c.mainBusinessLines.length > 0) {
        var lineSection = el('div', 'prh-haku-section');
        var lineTitle = el('div', 'prh-haku-section-title',
          t('detail.mainBusinessLine', translations) !== 'detail.mainBusinessLine'
            ? t('detail.mainBusinessLine', translations) : 'Paatoimiala');
        lineSection.appendChild(lineTitle);

        var lineDl = el('dl', 'prh-haku-dl');
        var line = c.mainBusinessLines[0];

        if (line.code) {
          var codeDt = el('dt', '', 'TOL');
          var codeDd = el('dd', '', line.code);
          lineDl.appendChild(codeDt);
          lineDl.appendChild(codeDd);
        }

        var lineDescDt = el('dt', '', t('detail.description', translations) !== 'detail.description' ? t('detail.description', translations) : 'Kuvaus');
        var lineDescDd = el('dd', '', getDescription(line.descriptions, 1));
        lineDl.appendChild(lineDescDt);
        lineDl.appendChild(lineDescDd);

        lineSection.appendChild(lineDl);
        wrapper.appendChild(lineSection);
      }

      // ── Addresses section ──
      if (c.addresses && c.addresses.length > 0) {
        var addrSection = el('div', 'prh-haku-section');
        var addrTitle = el('div', 'prh-haku-section-title',
          t('detail.addresses', translations) !== 'detail.addresses'
            ? t('detail.addresses', translations) : 'Osoitteet');
        addrSection.appendChild(addrTitle);

        c.addresses.forEach(function (addr) {
          var addrDiv = el('div', 'prh-haku-addr');
          var parts = [];
          if (addr.street) parts.push(addr.street);
          if (addr.buildingNumber) parts.push(addr.buildingNumber);
          if (addr.entrance) parts.push(addr.entrance);
          var line1 = parts.join(' ');
          var line2Parts = [];
          if (addr.postCode) line2Parts.push(addr.postCode);
          if (addr.city) line2Parts.push(addr.city);
          var line2 = line2Parts.join(' ');
          addrDiv.textContent = line1 + (line2 ? ', ' + line2 : '');
          if (addr.type !== undefined) {
            var typeLabel = el('span');
            typeLabel.style.cssText = 'font-size:0.8rem;color:var(--text-muted,#6b7280);margin-left:0.5rem;';
            typeLabel.textContent = '(' + (addr.type === 1 ? (t('detail.visitAddress', translations) !== 'detail.visitAddress' ? t('detail.visitAddress', translations) : 'Kayntiosoite') : (t('detail.postalAddress', translations) !== 'detail.postalAddress' ? t('detail.postalAddress', translations) : 'Postiosoite')) + ')';
            addrDiv.appendChild(typeLabel);
          }
          addrSection.appendChild(addrDiv);
        });
        wrapper.appendChild(addrSection);
      }

      // ── Registered entries section ──
      if (c.registeredEntries && c.registeredEntries.length > 0) {
        var regSection = el('div', 'prh-haku-section');
        var regTitle = el('div', 'prh-haku-section-title',
          t('detail.registeredEntries', translations) !== 'detail.registeredEntries'
            ? t('detail.registeredEntries', translations) : 'Rekisterimerkintoja');
        regSection.appendChild(regTitle);

        c.registeredEntries.forEach(function (entry) {
          var entryDiv = el('div', 'prh-haku-entry');
          var descDiv = el('div', 'prh-haku-entry-desc');
          var entryDesc = getDescription(entry.descriptions, 1);
          descDiv.textContent = entryDesc || entry.authority || '';
          entryDiv.appendChild(descDiv);

          if (entry.registrationDate || entry.endDate) {
            var dateDiv = el('div', 'prh-haku-entry-date');
            var dateParts = [];
            if (entry.registrationDate) dateParts.push(fmtDate(entry.registrationDate));
            if (entry.endDate) dateParts.push('- ' + fmtDate(entry.endDate));
            dateDiv.textContent = dateParts.join(' ');
            entryDiv.appendChild(dateDiv);
          }

          regSection.appendChild(entryDiv);
        });
        wrapper.appendChild(regSection);
      }

      // ── Company situations section ──
      if (c.companySituations && c.companySituations.length > 0) {
        var sitSection = el('div', 'prh-haku-section');
        var sitTitle = el('div', 'prh-haku-section-title',
          t('detail.companySituations', translations) !== 'detail.companySituations'
            ? t('detail.companySituations', translations) : 'Yritystilanteet');
        sitSection.appendChild(sitTitle);

        c.companySituations.forEach(function (sit) {
          var sitDiv = el('div', 'prh-haku-entry');
          var sitDesc = el('div', 'prh-haku-entry-desc');
          sitDesc.textContent = sit.type || '';
          sitDiv.appendChild(sitDesc);
          if (sit.registrationDate) {
            var sitDate = el('div', 'prh-haku-entry-date', fmtDate(sit.registrationDate));
            sitDiv.appendChild(sitDate);
          }
          sitSection.appendChild(sitDiv);
        });
        wrapper.appendChild(sitSection);
      }

      // ── Website ──
      if (c.contactDetails && c.contactDetails.length > 0) {
        var websites = c.contactDetails.filter(function (cd) {
          return cd.type && cd.type.toLowerCase() === 'www' || cd.contactType === 4;
        });
        if (websites.length > 0) {
          var webSection = el('div', 'prh-haku-section');
          var webTitle = el('div', 'prh-haku-section-title',
            t('detail.website', translations) !== 'detail.website'
              ? t('detail.website', translations) : 'Verkkosivu');
          webSection.appendChild(webTitle);

          websites.forEach(function (w) {
            var url = w.value || w.contact || '';
            if (url) {
              var link = el('a', 'prh-haku-link', url);
              link.href = url.indexOf('://') === -1 ? 'https://' + url : url;
              link.target = '_blank';
              link.rel = 'noopener noreferrer';
              webSection.appendChild(link);
            }
          });
          wrapper.appendChild(webSection);
        }
      }

      // ── Watchlist button ──
      var watchBtn = el('button', 'prh-haku-watchlist-btn');
      watchBtn.textContent = t('detail.addToWatchlist', translations) !== 'detail.addToWatchlist'
        ? t('detail.addToWatchlist', translations) : 'Lisaa seurantaan';
      watchBtn.addEventListener('click', async function () {
        watchBtn.disabled = true;
        try {
          await AIMEAT.prhYritysdata.addToWatchlist(dv(c.businessId), companyName);
          watchBtn.textContent = t('detail.addedToWatchlist', translations) !== 'detail.addedToWatchlist'
            ? t('detail.addedToWatchlist', translations) : 'Lisatty seurantaan';
          watchBtn.classList.add('prh-haku-added');
        } catch (e) {
          watchBtn.disabled = false;
          watchBtn.textContent = t('detail.watchlistError', translations) !== 'detail.watchlistError'
            ? t('detail.watchlistError', translations) : 'Virhe, yrita uudelleen';
        }
      });
      wrapper.appendChild(watchBtn);

      root.appendChild(wrapper);
    }

    // ── Actions ───────────────────────────────────────────────────────
    async function doSearch(query) {
      if (!query || !query.trim()) return;
      state.query = query.trim();
      state.loading = true;
      state.error = null;
      state.results = null;
      draw();

      try {
        var data = await AIMEAT.prhYritysdata.searchCompanies(state.query);
        // PRH API returns { results: [...] } or the array directly
        state.results = Array.isArray(data) ? data : (data && data.results ? data.results : []);
      } catch (e) {
        state.error = t('search.error', translations) !== 'search.error'
          ? t('search.error', translations)
          : 'Haku epaonnistui: ' + (e.message || 'Tuntematon virhe');
      }
      state.loading = false;
      draw();
    }

    async function loadCompanyDetail(businessId, name) {
      state.view = 'detail';
      state.loading = true;
      state.error = null;
      state.detail = null;
      draw();

      try {
        var data = await AIMEAT.prhYritysdata.getCompany(businessId);
        state.detail = data;
      } catch (e) {
        state.error = t('detail.error', translations) !== 'detail.error'
          ? t('detail.error', translations)
          : 'Yritystietojen lataus epaonnistui: ' + (e.message || 'Tuntematon virhe');
      }
      state.loading = false;
      draw();
    }
  }

  // ── Register ────────────────────────────────────────────────────────
  var exports = { render: render };
  if (AIMEAT.register) AIMEAT.register(LIB_NAME, exports);
  AIMEAT[LIB_NAME] = exports;
})(window.AIMEAT || (window.AIMEAT = {}));
